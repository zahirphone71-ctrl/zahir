// lib/scheduler.ts — جدولة المهام بدون تجميد
export async function yieldToMain(): Promise<void> {
  if ('scheduler' in globalThis && 'yield' in (globalThis as any).scheduler) {
    await (globalThis as any).scheduler.yield();
    return;
  }
  if (typeof MessageChannel !== 'undefined') {
    await new Promise<void>(resolve => {
      const ch = new MessageChannel();
      ch.port1.onmessage = () => { ch.port1.close(); resolve(); };
      ch.port2.postMessage(null);
    });
    return;
  }
  await new Promise(r => setTimeout(r, 0));
}

export async function processInChunks<T, R>(
  items: T[],
  processor: (item: T, index: number) => R | Promise<R>,
  opts: { budgetMs?: number; onProgress?: (done: number, total: number) => void; signal?: AbortSignal } = {}
): Promise<R[]> {
  const { budgetMs = 12, onProgress, signal } = opts;
  const results: R[] = [];
  let sliceStart = performance.now();
  for (let i = 0; i < items.length; i++) {
    if (signal?.aborted) throw new DOMException('Aborted', 'AbortError');
    results.push(await processor(items[i], i));
    if (performance.now() - sliceStart > budgetMs) {
      onProgress?.(i + 1, items.length);
      await yieldToMain();
      sliceStart = performance.now();
    }
  }
  onProgress?.(items.length, items.length);
  return results;
}

export function idleCallback(cb: () => void, timeout = 2000): void {
  if ('requestIdleCallback' in window) (window as any).requestIdleCallback(cb, { timeout });
  else setTimeout(cb, 1);
}

// lib/lruCache.ts
export class LRUCache<K, V> {
  private map = new Map<K, V>();
  constructor(private maxSize: number) {}
  get(key: K): V | undefined {
    const v = this.map.get(key);
    if (v !== undefined) { this.map.delete(key); this.map.set(key, v); }
    return v;
  }
  set(key: K, value: V): void {
    if (this.map.has(key)) this.map.delete(key);
    this.map.set(key, value);
    while (this.map.size > this.maxSize) {
      const k = this.map.keys().next().value;
      if (k !== undefined) this.map.delete(k);
    }
  }
  clear(): void { this.map.clear(); }
  get size(): number { return this.map.size; }
}

// lib/routerGuard.ts — حماية الراوتر
export interface GuardResult { ok: boolean; reason?: string; cpu?: number }

export class RouterGuard {
  private static failures = 0;
  private static circuitOpen = false;
  private static lastCheck = 0;
  private static cachedCpu = 0;

  static async canProceed(token: string, forceCheck = false): Promise<GuardResult> {
    if (this.circuitOpen) return { ok: false, reason: 'الدائرة مفتوحة مؤقتاً — الراوتر تحت ضغط سابق', cpu: this.cachedCpu };
    const now = Date.now();
    if (forceCheck || now - this.lastCheck > 10_000) {
      try {
        const res = await fetch('/api/vouchers/precheck', { headers: { 'X-Router-Token': token } });
        const data = await res.json();
        this.lastCheck = now;
        const cpu = data?.data?.routerInfo?.cpuLoad ?? 0;
        const freeRam = data?.data?.routerInfo?.freeMemory ?? Infinity;
        this.cachedCpu = cpu;
        if (cpu > 80) {
          this.circuitOpen = true;
          setTimeout(() => { this.circuitOpen = false; this.failures = 0; }, 30_000);
          return { ok: false, reason: `CPU الراوتر ${cpu}% — تم تأجيل الحقن تلقائياً`, cpu };
        }
        if (freeRam < 5 * 1024 * 1024) return { ok: false, reason: 'ذاكرة الراوتر منخفضة (< 5MB)', cpu };
      } catch {
        return { ok: true, cpu: this.cachedCpu };
      }
    }
    return { ok: true, cpu: this.cachedCpu };
  }

  static recordSuccess(): void { this.failures = 0; }
  static recordFailure(): void { this.failures++; if (this.failures >= 3) this.circuitOpen = true; }
  static getCpu(): number { return this.cachedCpu; }
  static reset(): void { this.failures = 0; this.circuitOpen = false; this.lastCheck = 0; this.cachedCpu = 0; }
}

// lib/abortableFetch.ts
import { useRef, useEffect, useCallback } from 'react';

export function useAbortableFetch() {
  const controllers = useRef<Set<AbortController>>(new Set());
  const abortableFetch = useCallback(async (input: RequestInfo, init?: RequestInit) => {
    const ctrl = new AbortController();
    controllers.current.add(ctrl);
    try { return await fetch(input, { ...init, signal: ctrl.signal }); }
    finally { controllers.current.delete(ctrl); }
  }, []);
  useEffect(() => () => { controllers.current.forEach(c => c.abort()); controllers.current.clear(); }, []);
  return abortableFetch;
}

// lib/regressionGuard.ts
export function startRegressionGuard(): void {
  if (typeof PerformanceObserver === 'undefined') return;
  try {
    new PerformanceObserver((list) => {
      for (const e of list.getEntries()) {
        if (e.duration > 50) console.warn(`[RegressionGuard] Long Task: ${e.duration.toFixed(0)}ms @ ${e.name}`);
      }
    }).observe({ entryTypes: ['longtask'] });
  } catch {}
}

      headers: { 'Content-Type': 'application/json', 'X-Router-Token': token },
      body: JSON.stringify({ ...params, quantity: chunkQty, batchId, offset: i }),
      signal,
    });
    const data = await res.json();
    if (!data.success) throw new Error(data.error);
    allVouchers.push(...data.vouchers);
    onChunkDone(data.vouchers, allVouchers.length, quantity);
    onProgress(Math.round((allVouchers.length / quantity) * 100), `تم ${allVouchers.length}/${quantity}`);
    await yieldToMain();
  }
  return { batchId, totalGenerated: allVouchers.length, vouchers: allVouchers };
}

// lib/injectionPipeline.ts
import { RouterGuard } from './routerGuard';

export async function injectVouchersPipeline(params: {
  vouchers: any[]; token: string; chunkSize?: number; delayBetweenMs?: number;
  onProgress: (done: number, total: number, cpu: number) => void;
  signal?: AbortSignal;
}): Promise<{ synced: number; failed: number }> {
  const { vouchers, token, chunkSize = 50, delayBetweenMs = 150, onProgress, signal } = params;
  let synced = 0, failed = 0;
  for (let i = 0; i < vouchers.length; i += chunkSize) {
    if (signal?.aborted) throw new DOMException('Aborted', 'AbortError');
    const guard = await RouterGuard.canProceed(token, i % (chunkSize * 5) === 0);
    if (!guard.ok) throw new Error(guard.reason);
    const chunk = vouchers.slice(i, i + chunkSize);
    try {
      const res = await fetch('/api/vouchers/sync', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'X-Router-Token': token },
        body: JSON.stringify({ voucherCodes: chunk.map(v => v.voucherCode) }),
        signal,
      });
      const data = await res.json();
      if (data.success) { synced += chunk.length; RouterGuard.recordSuccess(); }
      else { failed += chunk.length; RouterGuard.recordFailure(); }
    } catch { failed += chunk.length; RouterGuard.recordFailure(); }
    onProgress(synced, vouchers.length, RouterGuard.getCpu());
    await new Promise(r => setTimeout(r, delayBetweenMs));
  }
  return { synced, failed };
}

export function buildAutoLoginQrUrl(loginUrl: string, code: string, pin?: string, authMode?: string): string {
  const safeUrl = (loginUrl || '10.0.0.1').trim().replace(/^(https?:\/\/)+/i, '').replace(/[<>"'`\s]/g, '').slice(0, 200);
  let hostAndPath = safeUrl;

  const password = (authMode === 'user_pass' && pin) ? pin : (pin || code);
  const encodedUser = encodeURIComponent((code || '').slice(0, 64));
  const encodedPass = encodeURIComponent((password || '').slice(0, 64));

  const bw = Math.max(1, Math.round(preset.cardBorderWidth * fontScale));
  let border = 'none';
  if (preset.cardBorderStyle === 'double') border = `${Math.max(2, bw * 3)}px double ${design.borderColor || preset.borderColor}`;
  else if (preset.cardBorderStyle !== 'none') border = `${bw}px ${preset.cardBorderStyle} ${design.borderColor || preset.borderColor}`;
  return { background: preset.cardBgGradient || design.bgColor || preset.bgColor, color: design.cardTextColor || preset.cardTextColor, border, borderRadius: `${r}px`, boxShadow: preset.cardShadow, fontFamily: resolveFontStack(design.fontFamily || preset.fontFamily) };
}

export function computeQrSize(ctx: RenderContext): number { return Math.max(12, Math.round(ctx.qrSize * ctx.fontScale)); }

export function computePriceStyle(preset: ThemePreset, s: number): Record<string, string | number> {
  if (preset.priceStyle === 'badge') return { background: preset.primaryColor, color: '#ffffff', padding: `${Math.max(1, Math.round(2*s))}px ${Math.max(3, Math.round(8*s))}px`, borderRadius: '999px', fontWeight: 800 };
  if (preset.priceStyle === 'highlight') return { background: 'rgba(255,255,255,0.1)', padding: `${Math.max(1, Math.round(1*s))}px ${Math.max(2, Math.round(6*s))}px`, borderRadius: '4px', border: `1px solid ${preset.primaryColor}` };
  if (preset.priceStyle === 'strikethrough') return { textDecoration: 'line-through', textDecorationColor: preset.primaryColor, textDecorationThickness: `${Math.max(1, Math.round(2*s))}px` };
  return {};
}

export function computeElementZIndex(elKey: string, design: VoucherDesignState, baseZ = 10): number {
  const defs: Record<string, number> = { card: 0, network: 5, profile: 5, price: 5, time: 5, dataLimit: 5, user: 20, password: 20, qr: 15 };
  return (design.elements as any)?.[elKey]?.zIndex ?? (defs[elKey] ?? 0) + baseZ;
}

export function renderDecorations(preset: ThemePreset, scale: number): Array<{ style: React.CSSProperties; svg?: string }> {
  return (preset.decorations || []).map(id => DECORATION_DEFS[id]?.render(scale)).filter(Boolean) as any;
}

export function renderLogos(design: VoucherDesignState, scale: number): Array<{ id: string; src: string; style: React.CSSProperties }> {
  if (!design.logos?.length) return [];
  return design.logos.filter(l => l.visible !== false).map(l => ({ id: l.id, src: l.dataUrl, style: { position: 'absolute', left: `${l.x}%`, top: `${l.y}%`, width: `${Math.max(8, Math.round(l.size * scale))}px`, height: 'auto', transform: `translate(-50%,-50%) rotate(${l.rotation || 0}deg)`, opacity: l.opacity ?? 1, zIndex: l.zIndex ?? 5, pointerEvents: 'none' } }));
}

export function getAnimationForElement(preset: ThemePreset, elKey: string): { animation: string; delay: string } | null {
  const anim = THEME_ANIMATIONS[preset.animation];

ts
import QRCode from 'qrcode';
import { LRUCache } from '../../lib/lruCache';
import { processInChunks } from '../../lib/scheduler';
import { buildAutoLoginQrUrl } from './qrUtils';

const QR_CACHE = new LRUCache<string, string>(500);

export async function preGenerateQrCodes(vouchers: any[], loginUrl: string, authMode: string, qrSize = 120, onProgress?: (d: number, t: number) => void, signal?: AbortSignal): Promise<Map<string, string>> {
  const result = new Map<string, string>();
  await processInChunks(vouchers, async (v) => {
    if (signal?.aborted) throw new DOMException('Aborted', 'AbortError');
    if (!v.voucherCode) return;
    const key = `${v.voucherCode}|${v.voucherPin || ''}|${loginUrl}|${authMode}|${qrSize}`;
    const cached = QR_CACHE.get(key);
    if (cached) { result.set(v.voucherCode, cached); return; }
    try {
      const url = buildAutoLoginQrUrl(loginUrl, v.voucherCode, v.voucherPin, authMode);
      const dataUrl = await QRCode.toDataURL(url, { width: qrSize, margin: 1, errorCorrectionLevel: 'L', color: { dark: '#000000', light: '#ffffff' } });
      QR_CACHE.set(key, dataUrl); result.set(v.voucherCode, dataUrl);
    } catch { result.set(v.voucherCode, generateFallbackImage(qrSize)); }
  }, { budgetMs: 12, onProgress });
  return result;
}

function generateFallbackImage(size: number): string {
  if (typeof document === 'undefined') return '';
  const c = document.createElement('canvas'); c.width = c.height = size;
  const ctx = c.getContext('2d'); if (!ctx) return '';
  ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, size, size);
  ctx.fillStyle = '#94a3b8'; ctx.font = `${Math.max(8, size/6)}px sans-serif`; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; ctx.fillText('QR', size/2, size/2);
  const out = c.toDataURL('image/png'); c.width = c.height = 1;
  return out;
}

export function clearQrCache(): void { QR_CACHE.clear(); }

// lib/security.ts — تحقق من المدخلات
export const Validators = Object.freeze({
  code: (v: string) => /^[A-Z0-9]{4,16}$/.test(v) || 'الكود: 4-16 حرف/رقم',
  price: (v: number) => v >= 0 && v <= 999999 || 'السعر: 0-999999',
  quantity: (v: number) => v >= 1 && v <= 10000 || 'الكمية: 1-10000',
  url: (v: string) => /^[\w.-]+(:\d+)?(\/.*)?$/.test(v) || 'رابط غير صالح',
  ip: (v: string) => /^(\d{1,3}\.){3}\d{1,3}$/.test(v) || 'IP غير صالح',
  text: (v: string, max = 200) => v.length <= max || `النص: max ${max}`,
  number: (v: number, min: number, max: number) => v >= min && v <= max || `${min}-${max}`,
});

export function validateAll(checks: Array<() => true | string>): string[] {
  return checks.map(c => c()).filter(r => r !== true) as string[];
}

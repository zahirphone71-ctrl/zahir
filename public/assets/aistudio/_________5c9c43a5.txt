text
[UI Modal] → onSave(payload) → [الأب المجهول] → ؟؟؟ → [الراوتر]
المجهول هو "؟؟؟" — هذه هي الفجوة التي تجعل كل شيء وهميًا. التصحيح يملأ هذه الفجوة بطبقات محددة.

text

    ↓

    ↓
[Backend: RouterProfileController] → [RouterOSExecutor] → [RouterOS API]
    ↓
[Response] → [ErrorTranslator] → [UI Toast]
كل طبقة لها ملف محدد. لا قفزات. لا "سحر".

import { RouterOSAPI } from 'node-routeros';

export interface RouterConnectionConfig {
  host: string;
  port: number;
  username: string;
  password: string;
  timeout?: number;
}

export class RouterOSExecutor {
  private conn: RouterOSAPI;
  private connected = false;

  constructor(config: RouterConnectionConfig) {
    this.conn = new RouterOSAPI({
      host: config.host,
      user: config.username,
      password: config.password,
      port: config.port || 8728,
      timeout: config.timeout || 10000,
    });
  }

  async connect(): Promise<void> {
    if (this.connected) return;
    await this.conn.connect();
    this.connected = true;
  }

  async disconnect(): Promise<void> {
    if (this.connected) {
      await this.conn.close();
      this.connected = false;
    }
  }

  // ═══ إضافة بروفايل Hotspot ═══
  async addHotspotProfile(data: Record<string, string>): Promise<void> {
    await this.connect();
    await this.conn.write('/ip/hotspot/user/profile/add', data);
  }

  // ═══ تعديل بروفايل Hotspot ═══
  async updateHotspotProfile(oldName: string, data: Record<string, string>): Promise<void> {
    await this.connect();
    const found = await this.conn.write('/ip/hotspot/user/profile/print', {
      '?name': oldName,
    });
    if (!found.length) throw new Error(`البروفايل (${oldName}) غير موجود على الراوتر`);
    const id = found[0]['.id'];
    await this.conn.write('/ip/hotspot/user/profile/set', { '.id': id, ...data });
  }

  // ═══ حذف بروفايل Hotspot ═══
  async deleteHotspotProfile(name: string): Promise<void> {
    await this.connect();
    const found = await this.conn.write('/ip/hotspot/user/profile/print', { '?name': name });
    if (!found.length) throw new Error(`البروفايل (${name}) غير موجود على الراوتر`);
    await this.conn.write('/ip/hotspot/user/profile/remove', { '.id': found[0]['.id'] });
  }

  // ═══ قراءة بروفايلات Hotspot ═══
  async listHotspotProfiles(): Promise<Record<string, string>[]> {
    await this.connect();
    return await this.conn.write('/ip/hotspot/user/profile/print');
  }

  // ═══ PPP Profile (نفس النمط) ═══
  async addPppProfile(data: Record<string, string>): Promise<void> {
    await this.connect();
    await this.conn.write('/ppp/profile/add', data);
  }

  async updatePppProfile(oldName: string, data: Record<string, string>): Promise<void> {
    await this.connect();
    const found = await this.conn.write('/ppp/profile/print', { '?name': oldName });
    if (!found.length) throw new Error(`البروفايل (${oldName}) غير موجود`);
    await this.conn.write('/ppp/profile/set', { '.id': found[0]['.id'], ...data });
  }

  async deletePppProfile(name: string): Promise<void> {
    await this.connect();
    const found = await this.conn.write('/ppp/profile/print', { '?name': name });
    if (!found.length) throw new Error(`البروفايل (${name}) غير موجود`);
    await this.conn.write('/ppp/profile/remove', { '.id': found[0]['.id'] });
  }

  async listPppProfiles(): Promise<Record<string, string>[]> {
    await this.connect();
    return await this.conn.write('/ppp/profile/print');
  }

  // ═══ Userman Profile + Limitation (متعدد الخطوات) ═══
  async addUsermanProfileWithLimitation(profile: Record<string, string>, limitation: Record<string, string>): Promise<void> {
    await this.connect();
    // 1. إنشاء limitation
    await this.conn.write('/user-manager/limitation/add', limitation);
    // 2. إنشاء profile
    await this.conn.write('/user-manager/profile/add', profile);
    // 3. ربطهما
    await this.conn.write('/user-manager/profile-limitation/add', {
      profile: profile.name,
      limitation: limitation.name,
    });
  }

  async deleteUsermanProfile(name: string): Promise<void> {
    await this.connect();
    const found = await this.conn.write('/user-manager/profile/print', { '?name': name });
    if (!found.length) throw new Error(`البروفايل (${name}) غير موجود`);
    // 1. حذف الربط
    const links = await this.conn.write('/user-manager/profile-limitation/print', {
      '?profile': name,
    });
    for (const link of links) {
      await this.conn.write('/user-manager/profile-limitation/remove', { '.id': link['.id'] });
    }
    // 2. حذف البروفايل
    await this.conn.write('/user-manager/profile/remove', { '.id': found[0]['.id'] });
  }

  // ═══ فحص صحة الراوتر قبل العمليات ═══
  async getHealth(): Promise<{ cpuLoad: number; freeMemory: number }> {
    await this.connect();
    const resources = await this.conn.write('/system/resource/print');
    const r = resources[0] || {};
    return {
      cpuLoad: parseInt(r['cpu-load'] || '0', 10),
      freeMemory: parseInt(r['free-memory'] || '0', 10),
    };
  }
}

2. backend/lib/routeros/RouterGuard.ts (جديد)
الغرض: حماية الراوتر من التحميل الزائد — يمنع التنفيذ إذا كان CPU > 80% أو الذاكرة < 5MB.

ts
export class RouterGuard {
  private static circuitOpen = new Map<string, boolean>();
  private static failures = new Map<string, number>();

  static async canProceed(executor: RouterOSExecutor, routerId: string): Promise<{ ok: boolean; reason?: string }> {
    if (this.circuitOpen.get(routerId)) {
      return { ok: false, reason: 'الدائرة مفتوحة مؤقتًا — الراوتر تحت ضغط سابق' };
    }

    try {
      const health = await executor.getHealth();
      if (health.cpuLoad > 80) {
        this.openCircuit(routerId);
        return { ok: false, reason: `CPU الراوتر ${health.cpuLoad}% — تم تأجيل العملية` };
      }
      if (health.freeMemory < 5 * 1024 * 1024) {
        return { ok: false, reason: 'ذاكرة الراوتر منخفضة (< 5MB)' };
      }
      return { ok: true };
    } catch (err) {
      return { ok: false, reason: 'فشل الاتصال بالراوتر للفحص' };
    }
  }

  private static openCircuit(routerId: string): void {
    this.circuitOpen.set(routerId, true);
    setTimeout(() => this.circuitOpen.set(routerId, false), 30_000);
  }

  static recordSuccess(routerId: string): void {
    this.failures.set(routerId, 0);
  }

  static recordFailure(routerId: string): void {
    const f = (this.failures.get(routerId) || 0) + 1;
    this.failures.set(routerId, f);
    if (f >= 3) this.openCircuit(routerId);
  }
}

export class ErrorTranslator {
  static translate(error: any): string {
    const msg = String(error?.message || error || '').toLowerCase();

    if (msg.includes('already have') || msg.includes('already exists')) return 'البروفايل موجود مسبقًا بنفس الاسم';
    if (msg.includes('no such item') || msg.includes('not found')) return 'البروفايل غير موجود على الراوتر';
    if (msg.includes('invalid')) return 'قيمة غير صالحة في أحد الحقول — تحقق من صيغة السرعة والمهلات';
    if (msg.includes('timeout')) return 'انتهت مهلة الاتصال بالراوتر — تحقق من الشبكة';
    if (msg.includes('permission') || msg.includes('denied')) return 'لا تملك صلاحية لهذه العملية على الراوتر';
    if (msg.includes('connect') || msg.includes('econnrefused')) return 'فشل الاتصال بالراوتر — تحقق من العنوان والمنفذ';
    if (msg.includes('cannot remove') || msg.includes('in use')) return 'لا يمكن حذف البروفايل — هناك مستخدمون مرتبطون به';
    if (msg.includes('cpu') || msg.includes('overload')) return 'الراوتر تحت ضغط عالٍ — أعد المحاولة لاحقًا';

    return `خطأ من الراوتر: ${error?.message || 'غير معروف'}`;
  }
}

import { RouterOSExecutor } from '../lib/routeros/RouterOSExecutor';
import { RouterGuard } from '../lib/routeros/RouterGuard';
import { CliBuilder } from '../lib/routeros/CliBuilder';
import { ErrorTranslator } from '../lib/routeros/ErrorTranslator';

export class RouterProfileController {
  static async create(req, res) {
    const { serviceType, profileData } = req.body;
    const routerConfig = await getRouterConfig(req.user.id);

    const executor = new RouterOSExecutor(routerConfig);
    try {
      // 1. فحص صحة الراوتر
      const guard = await RouterGuard.canProceed(executor, routerConfig.id);
      if (!guard.ok) {
        return res.status(429).json({ success: false, message: guard.reason });
      }

      // 2. بناء payload حسب النوع
      if (serviceType === 'hotspot') {
        const payload = CliBuilder.buildHotspotProfilePayload(profileData);
        await executor.addHotspotProfile(payload);
      } else if (serviceType === 'pppoe') {
        const payload = CliBuilder.buildPppProfilePayload(profileData);
        await executor.addPppProfile(payload);
      } else if (serviceType === 'userman') {
        const { profile, limitation } = CliBuilder.buildUsermanPayloads(profileData);
        await executor.addUsermanProfileWithLimitation(profile, limitation);
        const link = CliBuilder.buildUsermanLinkPayload(profileData);
        await executor.conn.write('/user-manager/profile-limitation/add', link);
      } else {
        return res.status(400).json({ success: false, message: 'نوع خدمة غير معروف' });
      }

      // 3. حفظ في DB المحلي (مزامنة)
      await db.profiles.insert({ ...profileData, serviceType, routerId: routerConfig.id });

      // 4. تسجيل في Audit Log
      await db.auditLog.insert({
        userId: req.user.id,
        action: 'CREATE_PROFILE',
        serviceType,
        profileName: profileData.name,
        timestamp: new Date(),
      });

      RouterGuard.recordSuccess(routerConfig.id);
      res.json({ success: true, message: 'تم إنشاء البروفايل على الراوتر بنجاح' });
    } catch (error) {
      RouterGuard.recordFailure(routerConfig.id);
      res.status(500).json({ success: false, message: ErrorTranslator.translate(error) });
    } finally {
      await executor.disconnect();
    }
  }

  static async update(req, res) {
    const { serviceType, oldName, profileData } = req.body;
    const routerConfig = await getRouterConfig(req.user.id);
    const executor = new RouterOSExecutor(routerConfig);

    try {
      const guard = await RouterGuard.canProceed(executor, routerConfig.id);
      if (!guard.ok) return res.status(429).json({ success: false, message: guard.reason });

      if (serviceType === 'hotspot') {
        const payload = CliBuilder.buildHotspotProfilePayload(profileData);
        await executor.updateHotspotProfile(oldName, payload);
      } else if (serviceType === 'pppoe') {
        const payload = CliBuilder.buildPppProfilePayload(profileData);
        await executor.updatePppProfile(oldName, payload);
      } else if (serviceType === 'userman') {
        // Userman: حذف + إعادة إنشاء (لأن التعديل معقد)
        await executor.deleteUsermanProfile(oldName);
        const { profile, limitation } = CliBuilder.buildUsermanPayloads(profileData);
        await executor.addUsermanProfileWithLimitation(profile, limitation);
      }

      await db.profiles.update({ oldName }, { ...profileData, serviceType });
      await db.auditLog.insert({ userId: req.user.id, action: 'UPDATE_PROFILE', serviceType, profileName: profileData.name });

      RouterGuard.recordSuccess(routerConfig.id);
      res.json({ success: true, message: 'تم تعديل البروفايل على الراوتر بنجاح' });
    } catch (error) {
      RouterGuard.recordFailure(routerConfig.id);
      res.status(500).json({ success: false, message: ErrorTranslator.translate(error) });
    } finally {
      await executor.disconnect();
    }
  }

  static async delete(req, res) {
    const { serviceType, profileName } = req.body;
    const routerConfig = await getRouterConfig(req.user.id);
    const executor = new RouterOSExecutor(routerConfig);

    try {
      const guard = await RouterGuard.canProceed(executor, routerConfig.id);
      if (!guard.ok) return res.status(429).json({ success: false, message: guard.reason });

      // ✅ منع حذف default
      if (profileName === 'default') {
        return res.status(400).json({ success: false, message: 'لا يمكن حذف البروفايل الافتراضي' });
      }

      if (serviceType === 'hotspot') await executor.deleteHotspotProfile(profileName);
      else if (serviceType === 'pppoe') await executor.deletePppProfile(profileName);
      else if (serviceType === 'userman') await executor.deleteUsermanProfile(profileName);

      await db.profiles.delete({ name: profileName, serviceType });
      await db.auditLog.insert({ userId: req.user.id, action: 'DELETE_PROFILE', serviceType, profileName });

      RouterGuard.recordSuccess(routerConfig.id);
      res.json({ success: true, message: 'تم حذف البروفايل من الراوتر بنجاح' });
    } catch (error) {
      RouterGuard.recordFailure(routerConfig.id);
      res.status(500).json({ success: false, message: ErrorTranslator.translate(error) });
    } finally {

export class RouterConnectionPool {
  private static pools = new Map<string, RouterOSExecutor>();

  static get(routerConfig: RouterConnectionConfig): RouterOSExecutor {
    const key = `${routerConfig.host}:${routerConfig.port}`;
    if (!this.pools.has(key)) {
      this.pools.set(key, new RouterOSExecutor(routerConfig));
    }
    return this.pools.get(key)!;
  }

  static async releaseAll(): Promise<void> {
    for (const executor of this.pools.values()) {
      await executor.disconnect();
    }
    this.pools.clear();
  }
}

import { useState, useCallback } from 'react';

export function useProfileMutation() {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const mutate = useCallback(async (
    method: 'POST' | 'PUT' | 'DELETE',
    payload: any,
    signal?: AbortSignal
  ): Promise<{ success: boolean; message?: string }> => {
    setLoading(true);
    setError(null);
    try {
      const res = await fetch('/api/router/profiles', {
        method,
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload),
        signal,
        credentials: 'include',
      });
      const data = await res.json();
      if (!data.success) {
        setError(data.message || 'فشلت العملية');
        return data;
      }
      return data;
    } catch (err: any) {
      if (err.name === 'AbortError') return { success: false, message: 'أُلغيت العملية' };
      setError(err.message);
      return { success: false, message: err.message };
    } finally {
      setLoading(false);
    }
  }, []);

  return { mutate, loading, error };
}

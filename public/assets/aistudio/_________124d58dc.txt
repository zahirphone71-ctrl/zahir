import { RouterOSAPI } from 'node-routeros';

export interface RouterConfig {
  host: string; port: number; username: string; password: string; timeout?: number;
}

export class RouterOSExecutor {
  private conn: RouterOSAPI;
  private connected = false;

  constructor(config: RouterConfig) {
    this.conn = new RouterOSAPI({
      host: config.host,
      user: config.username,
      password: config.password,
      port: config.port || 8728,
      timeout: config.timeout || 15000,
    });
  }

  async connect(): Promise<void> {
    if (this.connected) return;
    await this.conn.connect();
    this.connected = true;
  }

  async disconnect(): Promise<void> {
    if (this.connected) { await this.conn.close(); this.connected = false; }
  }

  // ═══ فحص صحة الراوتر ═══
  async getHealth(): Promise<{ cpuLoad: number; freeMemory: number; uptime: string }> {
    await this.connect();
    const resources = await this.conn.write('/system/resource/print');
    const r = resources[0] || {};
    return {
      cpuLoad: parseInt(r['cpu-load'] || '0', 10),
      freeMemory: parseInt(r['free-memory'] || '0', 10),
      uptime: r['uptime'] || '0s',
    };
  }

  // ═══ قراءة المشتركين حسب النوع ═══
  async listHotspotUsers(): Promise<Record<string, string>[]> {
    await this.connect();
    return await this.conn.write('/ip/hotspot/user/print');
  }

  async listPppSecrets(): Promise<Record<string, string>[]> {
    await this.connect();
    return await this.conn.write('/ppp/secret/print');
  }

  async listUsermanUsers(): Promise<Record<string, string>[]> {
    await this.connect();
    return await this.conn.write('/user-manager/user/print');
  }

  async listHotspotActive(): Promise<Record<string, string>[]> {
    await this.connect();
    return await this.conn.write('/ip/hotspot/active/print');
  }

  async listPppActive(): Promise<Record<string, string>[]> {
    await this.connect();
    return await this.conn.write('/ppp/active/print');
  }

  async listUsermanActive(): Promise<Record<string, string>[]> {
    await this.connect();
    return await this.conn.write('/user-manager/session/print');
  }

  // ═══ إجراءات على المشتركين ═══
  async updateHotspotUser(name: string, data: Record<string, string>): Promise<void> {
    await this.connect();
    const found = await this.conn.write('/ip/hotspot/user/print', { '?name': name });
    if (!found.length) throw new Error(`المستخدم (${name}) غير موجود`);
    await this.conn.write('/ip/hotspot/user/set', { '.id': found[0]['.id'], ...data });
  }

  async deleteHotspotUser(name: string): Promise<void> {
    await this.connect();
    const found = await this.conn.write('/ip/hotspot/user/print', { '?name': name });
    if (!found.length) throw new Error(`المستخدم (${name}) غير موجود`);
    await this.conn.write('/ip/hotspot/user/remove', { '.id': found[0]['.id'] });
  }

  async disconnectActiveUser(serviceType: string, name: string): Promise<void> {
    await this.connect();
    const path = serviceType === 'hotspot' ? '/ip/hotspot/active' : serviceType === 'pppoe' ? '/ppp/active' : '/user-manager/session';
    const found = await this.conn.write(`${path}/print`, { '?user': name });
    if (!found.length) throw new Error('لا توجد جلسة نشطة');
    await this.conn.write(`${path}/remove`, { '.id': found[0]['.id'] });
  }

  async resetHotspotUserCounters(name: string): Promise<void> {
    await this.connect();
    const found = await this.conn.write('/ip/hotspot/user/print', { '?name': name });
    if (!found.length) throw new Error(`المستخدم (${name}) غير موجود`);
    await this.conn.write('/ip/hotspot/user/reset-counters', { '.id': found[0]['.id'] });
  }

  async toggleHotspotUserDisabled(name: string, disabled: boolean): Promise<void> {
    await this.connect();
    const found = await this.conn.write('/ip/hotspot/user/print', { '?name': name });
    if (!found.length) throw new Error(`المستخدم (${name}) غير موجود`);
    await this.conn.write('/ip/hotspot/user/set', { '.id': found[0]['.id'], disabled: disabled ? 'yes' : 'no' });
  }

  async listHotspotProfiles(): Promise<Record<string, string>[]> {
    await this.connect();
    return await this.conn.write('/ip/hotspot/user/profile/print');
  }

  // ═══ مراقبة CPU أثناء العمليات ═══
  async waitForHealth(maxCpu = 80, timeoutMs = 30000): Promise<void> {
    const start = Date.now();
    while (Date.now() - start < timeoutMs) {
      const health = await this.getHealth();
      if (health.cpuLoad < maxCpu) return;
      await new Promise((r) => setTimeout(r, 1000));
    }
    throw new Error('الراوتر تحت ضغط عالٍ — أعد المحاولة لاحقًا');
  }
}
2. backend/lib/routeros/RouterGuard.ts (جديد)
ts
export class RouterGuard {
  private static circuitOpen = new Map<string, boolean>();
  private static failures = new Map<string, number>();

  static async canProceed(executor: RouterOSExecutor, routerId: string): Promise<{ ok: boolean; reason?: string }> {
    if (this.circuitOpen.get(routerId)) {
      return { ok: false, reason: 'الدائرة مفتوحة مؤقتًا — الراوتر تحت ضغط سابق' };
    }
    try {
      const health = await executor.getHealth();
      if (health.cpuLoad > 80) {
        this.openCircuit(routerId);
        return { ok: false, reason: `CPU الراوتر ${health.cpuLoad}% — تم تأجيل العملية` };
      }
      if (health.freeMemory < 5 * 1024 * 1024) {
        return { ok: false, reason: 'ذاكرة الراوتر منخفضة (< 5MB)' };
      }
      return { ok: true };
    } catch {
      return { ok: false, reason: 'فشل الاتصال بالراوتر' };
    }
  }

  private static openCircuit(routerId: string): void {
    this.circuitOpen.set(routerId, true);
    setTimeout(() => this.circuitOpen.set(routerId, false), 30_000);
  }

  static recordSuccess(routerId: string): void { this.failures.set(routerId, 0); }
  static recordFailure(routerId: string): void {
    const f = (this.failures.get(routerId) || 0) + 1;
    this.failures.set(routerId, f);
    if (f >= 3) this.openCircuit(routerId);
  }
}

export class SubscribersController {
  // ═══ قراءة المشتركين الحقيقيين من الراوتر ═══
  static async list(req, res) {
    const { system, routerId } = req.query;
    const config = await getRouterConfig(routerId || req.user.defaultRouterId);
    const executor = new RouterOSExecutor(config);
    try {
      const guard = await RouterGuard.canProceed(executor, config.id);
      if (!guard.ok) return res.status(429).json({ success: false, message: guard.reason });

      let users: any[] = [];
      let active: any[] = [];

      if (system === 'hotspot') {
        users = await executor.listHotspotUsers();
        active = await executor.listHotspotActive();
      } else if (system === 'pppoe') {
        users = await executor.listPppSecrets();
        active = await executor.listPppActive();
      } else if (system === 'userman') {
        users = await executor.listUsermanUsers();
        active = await executor.listUsermanActive();
      }

      RouterGuard.recordSuccess(config.id);
      res.json({ success: true, users, active, fetchedAt: new Date().toISOString() });
    } catch (err: any) {
      RouterGuard.recordFailure(config.id);
      res.status(500).json({ success: false, message: ErrorTranslator.translate(err) });
    } finally {
      await executor.disconnect();
    }
  }

  // ═══ إجراء على مشترك ═══
  static async action(req, res) {
    const { system, routerId, action, username, payload } = req.body;
    const config = await getRouterConfig(routerId || req.user.defaultRouterId);
    const executor = new RouterOSExecutor(config);
    try {
      const guard = await RouterGuard.canProceed(executor, config.id);
      if (!guard.ok) return res.status(429).json({ success: false, message: guard.reason });

      if (action === 'delete') {
        if (system === 'hotspot') await executor.deleteHotspotUser(username);
        else if (system === 'pppoe') await executor.deletePppSecret(username);
        else if (system === 'userman') await executor.deleteUsermanUser(username);
      } else if (action === 'disconnect') {
        await executor.disconnectActiveUser(system, username);
      } else if (action === 'reset-counters') {
        await executor.resetHotspotUserCounters(username);
      } else if (action === 'toggle-suspend') {
        await executor.toggleHotspotUserDisabled(username, payload.disabled);
      } else if (action === 'update') {
        if (system === 'hotspot') await executor.updateHotspotUser(username, payload);
      } else {
        return res.status(400).json({ success: false, message: 'إجراء غير معروف' });
      }

      await db.auditLog.insert({ userId: req.user.id, action, system, username, timestamp: new Date() });
      RouterGuard.recordSuccess(config.id);
      res.json({ success: true, message: 'تم التنفيذ على الراوتر بنجاح' });
    } catch (err: any) {
      RouterGuard.recordFailure(config.id);
      res.status(500).json({ success: false, message: ErrorTranslator.translate(err) });
    } finally {
      await executor.disconnect();
    }
  }
}

import { useState, useEffect, useCallback, useRef } from 'react';

export function useRealSubscribers(system: 'hotspot' | 'pppoe' | 'userman', routerId?: string) {
  const [data, setData] = useState<{ users: any[]; active: any[] }>({ users: [], active: [] });
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [lastFetchedAt, setLastFetchedAt] = useState<string | null>(null);
  const abortRef = useRef<AbortController | null>(null);

  const fetchData = useCallback(async () => {
    abortRef.current?.abort();
    const ctrl = new AbortController();
    abortRef.current = ctrl;
    setLoading(true);
    setError(null);
    try {
      const params = new URLSearchParams({ system, ...(routerId ? { routerId } : {}) });
      const res = await fetch(`/api/subscribers?${params}`, { signal: ctrl.signal, credentials: 'include' });
      const json = await res.json();
      if (!json.success) { setError(json.message); return; }
      setData({ users: json.users || [], active: json.active || [] });
      setLastFetchedAt(json.fetchedAt);
    } catch (err: any) {
      if (err.name !== 'AbortError') setError(err.message);
    } finally {
      if (!ctrl.signal.aborted) setLoading(false);
    }
  }, [system, routerId]);

  useEffect(() => { fetchData(); return () => abortRef.current?.abort(); }, [fetchData]);

  const performAction = useCallback(async (action: string, username: string, payload?: any) => {
    try {
      const res = await fetch('/api/subscribers/action', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ system, routerId, action, username, payload }),
        credentials: 'include',
      });
      const json = await res.json();
      if (json.success) { await fetchData(); return { success: true }; }
      return { success: false, message: json.message };
    } catch (err: any) {
      return { success: false, message: err.message };
    }
  }, [system, routerId, fetchData]);

  return { data, loading, error, lastFetchedAt, refetch: fetchData, performAction };
}

export function filterColumnsByStatus(columns: TableColumnConfig[], statusFilter: string): TableColumnConfig[] {
  const REQUIRED: Record<string, string[]> = {
    idle: ['idleTime', 'disconnectTime'], online: ['connectionTime'], suspended: ['suspensionTime'],
    expired: ['remainingTimeAndData'], new: ['creationDate'],
  };
  const EXCLUDED: Record<string, string[]> = {
    idle: ['connectionTime', 'suspensionTime'], online: ['idleTime', 'disconnectTime', 'suspensionTime'],
    suspended: ['idleTime', 'connectionTime', 'disconnectTime'],
    expired: ['idleTime', 'connectionTime', 'suspensionTime'],
    new: ['connectionTime', 'disconnectTime', 'suspensionTime', 'idleTime'],
  };
  const required = REQUIRED[statusFilter] || [];
  const excluded = EXCLUDED[statusFilter] || [];

  const filtered = columns.filter((col) => {
    if (col.key === 'actions') return true;
    if (col.key === 'status') return statusFilter === 'all' || statusFilter === 'status';
    if (excluded.includes(col.key)) return false;
    return true;
  });

  const existingKeys = new Set(filtered.map((c) => c.key));
  required.forEach((key) => {
    if (!existingKeys.has(key)) {
      const col = ALL_SUBSCRIBER_COLUMNS.find((c) => c.key === key);
      if (col) filtered.push(col);
    }
  });
  return filtered;
}

import { ColumnCustomizerModal } from './ColumnCustomizerModal';

export const SubscribersPage = ({ system, routerId }: { system: 'hotspot' | 'pppoe' | 'userman'; routerId?: string }) => {
  const { data, loading, error, lastFetchedAt, refetch, performAction } = useRealSubscribers(system, routerId);
  const [statusFilter, setStatusFilter] = useState<StatusFilterType>('all');
  const [searchQuery, setSearchQuery] = useState('');
  const [searchColumn, setSearchColumn] = useState<SearchColumn>('all');
  const [selectedProfile, setSelectedProfile] = useState('all');
  const [density, setDensity] = useState<TableDensity>('normal');
  const [advancedFilters, setAdvancedFilters] = useState<AdvancedFilterState>(EMPTY_ADVANCED_FILTERS);
  const [isAdvancedOpen, setIsAdvancedOpen] = useState(false);
  const [isColumnCustomizerOpen, setIsColumnCustomizerOpen] = useState(false);
  const [visibleColumns, setVisibleColumns] = useState<string[]>(DEFAULT_COLUMN_KEYS);
  const [columnOrder, setColumnOrder] = useState<string[]>(DEFAULT_COLUMN_KEYS);
  const [selectedIds, setSelectedIds] = useState<string[]>([]);
  const [sortState, setSortState] = useState<ColumnSortState>({ key: null, direction: null });

  // ✅ تحويل البيانات الحقيقية من الراوتر إلى SubscriberRowItem
  const activeNames = useMemo(() => new Set(data.active.map(a => String(a.user || a.name || '').toLowerCase())), [data.active]);
  const items: SubscriberRowItem[] = useMemo(() => {
    return data.users.map(u => mapHotspotUserToRow(u, activeNames, []));
  }, [data.users, activeNames]);

  // ✅ فلترة + بحث + ترتيب
  const filtered = useMemo(() => {
    let result = items;
    // فلترة الحالة
    if (statusFilter !== 'all') result = result.filter(i => i.status === statusFilter);
    // بحث
    if (searchQuery.trim()) {
      const q = searchQuery.toLowerCase().trim();
      result = result.filter(i => {
        if (searchColumn === 'all') return i.name.toLowerCase().includes(q) || i.password?.toLowerCase().includes(q) || i.comment?.toLowerCase().includes(q);
        if (searchColumn === 'username') return i.name.toLowerCase().includes(q);
        if (searchColumn === 'mac') return i.macAddress?.toLowerCase().includes(q);
        if (searchColumn === 'profile') return i.profile.toLowerCase().includes(q);
        if (searchColumn === 'notes') return i.comment?.toLowerCase().includes(q);
        return true;
      });
    }
    // فلترة البروفايل
    if (selectedProfile !== 'all') result = result.filter(i => i.profile === selectedProfile);
    // فلترة متقدمة
    if (advancedFilters.activity !== 'all') result = result.filter(i => advancedFilters.activity === 'active' ? i.status === 'online' : advancedFilters.activity === 'inactive' ? i.status === 'idle' : i.status === 'suspended');
    // ترتيب
    if (sortState.key && sortState.direction) {
      result = [...result].sort((a, b) => {
        const av = (a as any)[sortState.key!];
        const bv = (b as any)[sortState.key!];
        const cmp = String(av || '').localeCompare(String(bv || ''));
        return sortState.direction === 'asc' ? cmp : -cmp;
      });
    }
    return result;
  }, [items, statusFilter, searchQuery, searchColumn, selectedProfile, advancedFilters, sortState]);

  // ✅ إجراءات حقيقية على المايكروتك
  const handleQuickDelete = async (item: SubscriberRowItem) => {
    const res = await performAction('delete', item.name);
    if (res.success) showToast(`تم حذف (${item.name}) من الراوتر`, 'success');
    else showToast(res.message || 'فشل الحذف', 'error');
  };

  const handleQuickDisconnect = async (item: SubscriberRowItem) => {
    const res = await performAction('disconnect', item.name);
    if (res.success) showToast(`تم فصل (${item.name})`, 'success');
    else showToast(res.message || 'فشل الفصل', 'error');
  };

  const handleQuickToggleSuspend = async (item: SubscriberRowItem) => {
    const res = await performAction('toggle-suspend', item.name, { disabled: !item.disabled });
    if (res.success) showToast(item.disabled ? 'تم تفعيل الحساب' : 'تم تعليق الحساب', 'success');
    else showToast(res.message || 'فشلت العملية', 'error');
  };

  return (
    <div className="space-y-4" dir="rtl">
      {loading && <div className="skeleton h-1 w-full rounded-full" />}
      {error && <div className="bg-rose-950/40 border border-rose-500/30 rounded-xl p-3 text-rose-300 text-xs">{error}</div>}
      <SubscribersFilterBar
        searchQuery={searchQuery} onSearchChange={setSearchQuery}
        searchColumn={searchColumn} onSearchColumnChange={setSearchColumn}
        selectedProfile={selectedProfile} onProfileChange={setSelectedProfile}
        profilesList={[]} statusFilter={statusFilter} onStatusFilterChange={setStatusFilter}
        statusCounts={computeStatusCounts(items)} onOpenColumnCustomizer={() => setIsColumnCustomizerOpen(true)}
        density={density} onChangeDensity={setDensity} activeSystemName={system}
        isAdvancedOpen={isAdvancedOpen} onToggleAdvanced={() => setIsAdvancedOpen(!isAdvancedOpen)}
        advancedFilters={advancedFilters}
      />
      {isAdvancedOpen && <AdvancedFilterPanel isOpen={isAdvancedOpen} onClose={() => setIsAdvancedOpen(false)} filters={advancedFilters} onChangeFilters={(updater) => setAdvancedFilters(updater)} onResetFilters={() => setAdvancedFilters(EMPTY_ADVANCED_FILTERS)} profilesList={[]} totalMatching={filtered.length} totalSubscribers={items.length} />}
      <SubscribersTable
        system={system} items={filtered} selectedIds={selectedIds}
        onToggleSelectAll={() => setSelectedIds(selectedIds.length === filtered.length ? [] : filtered.map(i => i.id))}
        onToggleSelectOne={(id) => setSelectedIds(prev => prev.includes(id) ? prev.filter(x => x !== id) : [...prev, id])}
        sortState={sortState} onSortChange={(key) => setSortState(prev => ({ key, direction: prev.key === key && prev.direction === 'asc' ? 'desc' : 'asc' }))}
        visibleColumns={visibleColumns} columnOrder={columnOrder} onChangeColumnOrder={setColumnOrder}
        density={density}
        onQuickDelete={handleQuickDelete}
        onQuickDisconnect={handleQuickDisconnect}
        onQuickToggleSuspend={handleQuickToggleSuspend}
        showToast={showToast}
        statusFilter={statusFilter}
      />
      {isColumnCustomizerOpen && <ColumnCustomizerModal isOpen={isColumnCustomizerOpen} onClose={() => setIsColumnCustomizerOpen(false)} columnOrder={columnOrder} visibleColumns={visibleColumns} onChangeOrder={setColumnOrder} onToggleColumn={(k) => setVisibleColumns(prev => prev.includes(k) ? prev.filter(x => x !== k) : [...prev, k])} onSetAllVisible={setVisibleColumns} onResetDefaults={() => { setVisibleColumns(DEFAULT_COLUMN_KEYS); setColumnOrder(DEFAULT_COLUMN_KEYS); }} density={density} onChangeDensity={setDensity} />}
    </div>
  );
};

function computeStatusCounts(items: SubscriberRowItem[]): StatusCounts {
  return {
    total: items.length,
    online: items.filter(i => i.status === 'online').length,
    idle: items.filter(i => i.status === 'idle').length,
    suspended: items.filter(i => i.status === 'suspended').length,
    expired: items.filter(i => i.status === 'expired').length,
    new: items.filter(i => i.status === 'new').length,
  };
}
